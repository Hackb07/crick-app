        /**
         * Check if innings is complete
         * @param {number} wickets - Current wicket count
         * @param {number} completedOvers - Number of completed overs (full overs)
         * @param {number} currentBalls - Current balls in the over (0-5)
         * @param {number} maxOvers - Maximum overs allowed
         * @returns {boolean} True if innings is complete
         */
        function isInningsComplete(wickets, completedOvers, currentBalls, maxOvers) {
            // Check wickets
            if (wickets >= 10) return true;
            
            // Calculate total overs (including partial over)
            // For example: 2 overs + 0 balls = 2.0 overs, which is >= 2.0 maxOvers
            const totalOvers = completedOvers + (currentBalls / 6);
            
            // Innings complete when total overs >= max overs
            // Use >= to handle exact match (e.g., 2.0 >= 2.0)
            if (totalOvers >= maxOvers) return true;
            
            // CRITICAL FIX: For innings 2, also check if target is reached
            // If batting team reaches or exceeds target, innings is complete (match won)
            if (currentInnings === 2 && firstInningsTotal > 0) {
                const target = firstInningsTotal + 1;
                if (currentScore >= target) {
                    return true; // Target reached - innings complete!
                }
            }
            
            return false;
        }
        
        function calculateOvers(balls) {
            const completedOvers = Math.floor(balls / 6);
            const remainingBalls = balls % 6;
            return completedOvers + (remainingBalls / 10);
        }
        
        function calculateRunRate(runs, balls) {
            if (balls === 0) return 0.0;
            return (runs / balls) * 6;
        }
        
        function calculateTarget(firstInningsTotal) {
            return firstInningsTotal + 1;
        }
        
        function calculateRequiredRunRate(target, currentScore, remainingOvers) {
            const remainingRuns = target - currentScore;
            if (remainingRuns <= 0 || remainingOvers <= 0) {
                return 0.0;
            }
            return remainingRuns / remainingOvers;
        }

        /**
         * Update new batsman dropdown to show only available players
         * Excludes dismissed players and current batsmen (striker/non-striker)
         */
        function updateNewBatsmanDropdown() {
            const dropdown = document.getElementById('new-batsman-select');
            if (!dropdown) return;
            
            // Get all batting team players from the striker dropdown (source of truth)
            const strikerSelect = document.getElementById('striker');
            if (!strikerSelect) return;
            
            const currentStrikerId = parseInt(strikerSelect.value) || null;
            const nonStrikerSelect = document.getElementById('non-striker');
            const currentNonStrikerId = parseInt(nonStrikerSelect?.value) || null;
            
            // Get dismissed players from event history (players who had wickets)
            const dismissedPlayerIds = new Set();
            eventHistory.forEach(event => {
                if (event.type === 'wicket' && event.striker_id) {
                    dismissedPlayerIds.add(event.striker_id);
                }
            });
            
            // Get all available options from striker dropdown
            const allPlayers = Array.from(strikerSelect.options).map(opt => ({
                value: opt.value,
                text: opt.text,
                playerId: parseInt(opt.value)
            })).filter(opt => opt.value); // Exclude empty option
            
            // Clear dropdown (except first empty option)
            dropdown.innerHTML = '<option value="">Select...</option>';
            
            // Add available players (exclude dismissed and current batsmen)
            allPlayers.forEach(player => {
                const playerId = player.playerId;
                
                // Skip if dismissed
                if (dismissedPlayerIds.has(playerId)) {
                    return;
                }
                
                // Skip if currently batting
                if (playerId === currentStrikerId || playerId === currentNonStrikerId) {
                    return;
                }
                
                // Add to dropdown
                const option = document.createElement('option');
                option.value = player.value;
                option.textContent = player.text;
                dropdown.appendChild(option);
            });
            
            // Update help text
            const helpText = dropdown.parentElement.querySelector('small');
            if (helpText && dropdown.options.length - 1 < allPlayers.length) {
                helpText.textContent = 'Showing available players only (excludes dismissed and current batsmen)';
            } else if (helpText) {
                helpText.textContent = 'New batsman will take the strike position';
            }
        }

        function generateUUID() {
            return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
                const r = Math.random() * 16 | 0;
                const v = c === 'x' ? r : (r & 0x3 | 0x8);
                return v.toString(16);
            });
        }

        /**
         * Record a ball event (run, wicket, or extra)
         * 
         * Validates required fields (striker, bowler) before recording.
         * Handles sequence conflicts by reloading match state.
         * Updates score, wickets, overs, and player statistics.
         * 
         * @param {Object} data - Ball data {type, runs?, extra_type?, dismissal_type?}
         */
        function recordBall(data) {
            const striker = document.getElementById('striker').value;
            const nonStriker = document.getElementById('non-striker').value;
            const bowler = document.getElementById('bowler').value;
            
            if (!striker || !bowler) {
                alert('Please select striker and bowler');
                return;
            }

            // Store event for undo
            const eventData = {
                type: data.type,
                runs: data.runs || 0,
                striker_id: parseInt(striker),
                bowler_id: parseInt(bowler),
                extra_type: data.extra_type || null,
                dismissal_type: data.dismissal_type || null,
                timestamp: Date.now()
            };
            eventHistory.push(eventData);

            // Create event payload
            const payload = {
                type: data.type,
                runs: data.runs || 0,
                striker_id: parseInt(striker),
                non_striker_id: nonStriker ? parseInt(nonStriker) : null,
                bowler_id: parseInt(bowler),
                extra_type: data.extra_type || null,
                dismissal_type: data.dismissal_type || null,
                innings: currentInnings
            };
            
            // Add new batsman ID for wickets
            if (data.type === 'wicket' && data.new_batsman_id) {
                payload.new_batsman_id = data.new_batsman_id;
            }
            
            // Add retired hurt info
            if (data.type === 'retired_hurt') {
                payload.retired_player_id = data.retired_player_id;
                payload.replacement_player_id = data.replacement_player_id;
            }
            
            const event = {
                event_uuid: generateUUID(),
                client_id: 'web-scorer-' + Date.now(),
                client_ts: new Date().toISOString(),
                client_base_seq: serverSeq,
                ball_index: currentBalls, // This will be updated by server for wide/no-ball
                appearance_id: null,
                payload_json: payload
            };

            // Send as batch (API expects batch format)
            fetch('<?= apiUrl('events.php') ?>', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                credentials: 'same-origin',
                body: JSON.stringify({
                    match_id: matchId,
                    client_base_seq: serverSeq,
                    events: [event]
                })
            })
            .then(response => {
                // Check if response is JSON
                const contentType = response.headers.get('content-type');
                if (!contentType || !contentType.includes('application/json')) {
                    return response.text().then(text => {
                        console.error('Non-JSON response from events API:', text.substring(0, 500));
                        throw new Error('Server returned HTML instead of JSON. Check server logs for PHP errors.');
                    });
                }
                return response.json();
            })
            .then(result => {
                if (result.success) {
                    serverSeq = result.server_seq;
                    clientSeq++;
                    
                    // Update stats
                    updateStatsAfterBall(data, striker, bowler);
                    
                    // Update score display
                    if (data.type === 'run') {
                        currentScore += data.runs || 0;
                        currentBalls++;
                        
                        // Update player stats
                        if (playerStats.batsmen[striker]) {
                            playerStats.batsmen[striker].runs += data.runs || 0;
                            playerStats.batsmen[striker].balls++;
                            if (data.runs === 4) playerStats.batsmen[striker].fours++;
                            if (data.runs === 6) playerStats.batsmen[striker].sixes++;
                        }
                        if (playerStats.bowlers[bowler]) {
                            playerStats.bowlers[bowler].runs += data.runs || 0;
                            playerStats.bowlers[bowler].balls++;
                        }
                        
                        // Handle strike rotation using helper function
                        // Odd runs (1, 3) rotate strike, even runs (0, 2, 4, 6) don't
                        if (shouldRotateStrike(data.runs || 0, false)) {
                            swapStrike();
                        }
                    } else if (data.type === 'wicket') {
                        currentWickets++;
                        const isLegalDelivery = true; // Wickets are always legal deliveries
                        currentBalls++;
                        
                        if (playerStats.batsmen[striker]) {
                            playerStats.batsmen[striker].balls++;
                        }
                        if (playerStats.bowlers[bowler]) {
                            playerStats.bowlers[bowler].wickets++;
                            playerStats.bowlers[bowler].balls++;
                        }
                        
                        // Add wicket to current over balls for display
                        currentOverBalls.push('W');
                        
                        // Update striker to new batsman (if provided) - after API success
                        if (data._newBatsmanId) {
                            const strikerSelect = document.getElementById('striker');
                            strikerSelect.value = data._newBatsmanId;
                            updateStriker();
                        }
                    } else if (data.type === 'retired_hurt') {
                        // Retired hurt doesn't count as wicket or ball
                        // Just replace the batsman
                        // No score/ball changes
                    } else if (data.type === 'extra') {
                        // Extras: wide, no-ball (1 run + additional runs 0,1,2,3,4,6), bye, leg-bye (1+ runs)
                        const extraRuns = data.runs || 1;
                        currentScore += extraRuns;
                        if (playerStats.bowlers[bowler]) {
                            playerStats.bowlers[bowler].runs += extraRuns;
                        }
                        
                        // Handle strike rotation for wide/no-ball with runs
                        // If wide/no-ball has additional runs (2,3,4,5,7), rotate strike based on additional runs
                        if ((data.extra_type === 'wide' || data.extra_type === 'no-ball') && extraRuns > 1) {
                            const additionalRuns = extraRuns - 1; // Subtract base run
                            if (shouldRotateStrike(additionalRuns, data.extra_type === 'no-ball')) {
                                swapStrike();
                            }
                        }
                        
                        // Use helper function to check if ball is legal
